<script>
(() => {
  const input = document.getElementById("quote-search");
  const rows = [...document.querySelectorAll(".quote-row")];
  const empty = document.getElementById("no-quote-results");

  if (!input || !rows.length) return;

  input.addEventListener("input", () => {
    const query = input.value.trim().toLowerCase();
    let visible = 0;

    rows.forEach((row) => {
      const matches = row.dataset.search
        .toLowerCase()
        .includes(query);

      row.hidden = !matches;

      if (matches) visible += 1;
    });

    if (empty) {
      empty.classList.toggle("hidden", visible !== 0);
    }
  });
})();
</script>
